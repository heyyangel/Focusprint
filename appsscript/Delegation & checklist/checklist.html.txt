  function parseDateTimeString(str) {
    if (!str) return null;
    let d = new Date(str);
    if (!isNaN(d.getTime())) return d;
    
    try {
      const parts = String(str).trim().split(/\s+/);
      if (parts.length >= 3) {
        const day = parseInt(parts[0]);
        const monthStr = parts[1];
        const yearStr = parts[2];
        
        const monthNames = {
          jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5,
          jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11
        };
        const month = monthNames[monthStr.toLowerCase().substring(0, 3)];
        if (month === undefined) return null;
        let year = parseInt(yearStr);
        if (year < 100) year += 2000;
        
        let hrs = 0, mins = 0, secs = 0;
        if (parts[3]) {
          const timeParts = parts[3].split(':');
          hrs = parseInt(timeParts[0] || 0);
          mins = parseInt(timeParts[1] || 0);
          secs = parseInt(timeParts[2] || 0);
        }
        return new Date(year, month, day, hrs, mins, secs);
      }
    } catch (e) {
      return null;
    }
    return null;
  }

  function getLocalISODate(d) {
    if (!d || isNaN(d.getTime())) return '';
    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${year}-${month}-${day}`;
  }

      function escapeHtml(s) {

      function formatUiDate(dateStr) {
        if (!dateStr) return '';
        const parts = String(dateStr).split('-');
        if (parts.length !== 3) return dateStr;
        const year = parts[0].substring(2); // '26'
        const monthIdx = parseInt(parts[1], 10) - 1;
        const day = parseInt(parts[2], 10);
        const monthsShort = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
        const monthName = monthsShort[monthIdx] || '';
        return `${day} ${monthName} ${year}`;
      }
      function getUpcomingDayDate(dayOfWeek) {
        const today = new Date();
        const result = new Date(today);
        const currentDay = today.getDay(); // 0 = Sunday, 1 = Monday, ..., 6 = Saturday
        let daysToAdd = (dayOfWeek - currentDay + 7) % 7;
        result.setDate(today.getDate() + daysToAdd);
        return result.toISOString().split('T')[0];
      }

      function applyRoleFilter(data) {
        // Ensure data is a safe array
        const safeData = Array.isArray(data) ? data : [];
        const normUser = (currentUser || '').trim().toLowerCase();
        
        return safeData.filter(r => {
          // Row details check
          const isVerifier = (r['Verified by'] || '').trim().toLowerCase() === normUser;
          const isDoer = (r['Name'] || '').trim().toLowerCase() === normUser;
          
          // Agar aage chal kar Checklist mein bhi Sub-Monitoring add hota hai, toh ye automatically kaam karega
          const isSubMonitor = (r['Sub Monitoring'] || '').trim().toLowerCase() === normUser;

          // 1. Agar normal user / doer hai, toh sirf wahi tasks dikhenge jisme wo Doer, Verifier, ya Monitor hai
          if (currentRole === 'Delegatee' || currentRole === 'User') {
            return isDoer || isVerifier || isSubMonitor;
          }

          // 2. Agar Manager / Admin / PC hai (Jinke paas tabs hote hain)
          if (currentRoleSwitcher === 'all') {
            return true; // Sab dikhega
          }
          if (currentRoleSwitcher === 'byme') {
            return isVerifier || isSubMonitor; // Jo usne verify/monitor karne hain
          }
          if (currentRoleSwitcher === 'tome') {
            return isDoer || isSubMonitor || isVerifier; // Jo usko khud karne hain
          }

          return false;
        });
      }

        if (!Array.isArray(visible)) visible = [];
        
        const summary = new Map();

          const name = row.Name; if (!name) continue;
          if (!summary.has(name)) summary.set(name, { groups: new Set(), tasks: 0 });

          summary.get(name).tasks++;

        const container = document.getElementById('summaryList');

      // ── Apply all filters ──────────────────────────────────────────────────
    function applyFiltersAndRender() {
  let data = applyRoleFilter([...checklistData]);

  // Sidebar filter
  if (currentNameFilter) {
    data = data.filter(r => r.Name === currentNameFilter);
  } else if (filterNames.length) {
    data = data.filter(r => filterNames.includes(r.Name));
  }

  if (filterFreqs.length) data = data.filter(r => filterFreqs.includes(r.Frequency));
  if (filterVerified.length) data = data.filter(r => filterVerified.includes((r['Verified by'] || '').trim()));

  const fromDate = document.getElementById('filterFromDate').value;
  const toDate = document.getElementById('filterToDate').value;
  
  const todayLocal = new Date();
  const todayStr = `${todayLocal.getFullYear()}-${String(todayLocal.getMonth() + 1).padStart(2, '0')}-${String(todayLocal.getDate()).padStart(2, '0')}`;

  
  if ((fromDate && fromDate > todayStr) || (toDate && toDate > todayStr)) {
      showToast("Future dates cannot be selected!", "error");
      return; 
  }

  if (fromDate || toDate) {
    data = data.filter(r => {
      if (!r['Due Date']) return false;
      const d = parseDateTimeString(r['Due Date']); // parse raw string safely
      if (!d) return false;
      
      const dateStr = getLocalISODate(d); // get timezone-safe local date string
      
      let match = true;
      if (fromDate) match = match && (dateStr >= fromDate);
      if (toDate) match = match && (dateStr <= toDate);
      return match;
    });
  } else if (currentQuickDate === 'upcoming') {
    // Keep your existing "Today & Tomorrow" logic here, but make it timezone safe!
    const today = new Date(); today.setHours(0,0,0,0);
    const tomorrow = new Date(); tomorrow.setDate(tomorrow.getDate() + 1); tomorrow.setHours(23,59,59,999);
    data = data.filter(r => {
        if (!r['Due Date']) return false;
        const d = parseDateTimeString(r['Due Date']);
        if (!d) return false;
        return d.getTime() >= today.getTime() && d.getTime() <= tomorrow.getTime();
    });
  }
  // Group and render...
  const groups = new Map();
  data.forEach(row => {
    const gid = row['Group ID'];
    if (gid) { if (!groups.has(gid)) groups.set(gid, []); groups.get(gid).push(row); }
  });
  filteredGroupsCache = groups;
  renderGroupedTable(groups);
}

      // ── Pagination ─────────────────────────────────────────────────────────
      function renderPagination(totalGroups, totalPages) {
        const pgBar = document.getElementById('paginationBar');
        if (totalPages <= 1) { pgBar.classList.add('hidden'); return; }
        pgBar.classList.remove('hidden');
        const start = (currentPage-1)*pageSize+1;
        const end = Math.min(currentPage*pageSize, totalGroups);
        document.getElementById('paginationInfo').textContent = `Showing ${start}–${end} of ${totalGroups} groups`;

        const pgNumbers = document.getElementById('pgNumbers');
        pgNumbers.innerHTML = '';
        let startPg = Math.max(1, currentPage-2), endPg = Math.min(totalPages, startPg+4);
        if (endPg-startPg < 4) startPg = Math.max(1, endPg-4);
        for (let p = startPg; p <= endPg; p++) {
          const btn = document.createElement('button');
          btn.textContent = p;
          btn.className = `w-7 h-7 rounded-lg text-xs font-semibold border transition ${p===currentPage?'bg-blue-600 text-white border-blue-600 shadow':'bg-white text-gray-600 border-gray-200 hover:bg-blue-50 hover:border-blue-300'}`;
          btn.addEventListener('click', () => { currentPage = p; renderGroupedTable(filteredGroupsCache); });
          pgNumbers.appendChild(btn);
        }
        const pgFirst = document.getElementById('pgFirst'), pgPrev = document.getElementById('pgPrev');
        const pgNext = document.getElementById('pgNext'), pgLast = document.getElementById('pgLast');
        pgFirst.disabled = pgPrev.disabled = currentPage === 1;
        pgNext.disabled = pgLast.disabled = currentPage === totalPages;
        pgFirst.onclick = () => { currentPage=1; renderGroupedTable(filteredGroupsCache); };
        pgPrev.onclick  = () => { if(currentPage>1){currentPage--; renderGroupedTable(filteredGroupsCache);} };
        pgNext.onclick  = () => { if(currentPage<totalPages){currentPage++; renderGroupedTable(filteredGroupsCache);} };
        pgLast.onclick  = () => { currentPage=totalPages; renderGroupedTable(filteredGroupsCache); };
      }

      function exportToCSV() {
        if (!checklistData.length) { showToast('No data to export','warning'); return; }
        const cols = ['Created At','Name','Group ID','Task ID','Task Description','Frequency','Due Date','Proof is Required','Verified by'];
        const rows = [cols.join(',')];
        checklistData.forEach(r => {
          rows.push(cols.map(c => {
            const v = r[c] != null ? String(r[c]) : '';
            return v.includes(',') || v.includes('"') || v.includes('\n') ? `"${v.replace(/"/g,'""')}"` : v;
          }).join(','));
        });
        const blob = new Blob([rows.join('\n')], { type: 'text/csv' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url; a.download = `checklist_export_${new Date().toISOString().slice(0,10)}.csv`;
        a.click(); URL.revokeObjectURL(url);
        showToast('Exported successfully');
      }

    function collectBulkRows() {
        const rows = [];
        document.querySelectorAll('#bulkTableBody tr.tr-inputs').forEach(trInputs => {
          const rowId = trInputs.getAttribute('data-row-id');
          const trDesc = document.querySelector(`#bulkTableBody tr.tr-description[data-row-id="${rowId}"]`);
          
          const name = trInputs.querySelector('.bulk-name')?.value.trim();
          const desc = trDesc?.querySelector('.bulk-desc')?.value.trim();
          if (!name || !desc) return;
          
          const email = trInputs.getAttribute('data-email') || '';
          const frequency = trInputs.querySelector('.bulk-freq')?.value || '';
          const proof = trInputs.querySelector('.bulk-proof')?.value || 'No';
          const verifiedBy = trInputs.querySelector('.bulk-verified')?.value || '';
          
          let dates = trInputs.getDates ? trInputs.getDates() : [];
          if (!dates.length) return;

          // Use the FIRST selected date as the starting point
          const startDate = new Date(dates[0]);

          if (frequency === 'Daily') {
            // Generate 365 days (1 complete year)
            for (let i = 0; i < 365; i++) {
              let d = new Date(startDate);
              d.setDate(d.getDate() + i);
              rows.push({ name, email, taskDescription: desc, frequency, dueDate: d.toISOString().split('T')[0], proofRequired: proof, verifiedBy });
            }
          } 
          else if (frequency === 'Weekly') {
            // Generate 52 weeks (1 complete year)
            for (let i = 0; i < 52; i++) {
              let d = new Date(startDate);
              d.setDate(d.getDate() + (i * 7));
              rows.push({ name, email, taskDescription: desc, frequency, dueDate: d.toISOString().split('T')[0], proofRequired: proof, verifiedBy });
            }
          } 
          else if (frequency === 'Monthly') {
            // Generate 12 months (1 complete year)
            for (let i = 0; i < 12; i++) {
              let d = new Date(startDate);
              let expectedMonth = (d.getMonth() + i) % 12;
              d.setMonth(d.getMonth() + i);
              
              // Handle end-of-month edge cases (e.g., Jan 31 -> Feb 28 -> Mar 31)
              if (d.getMonth() !== expectedMonth && d.getMonth() !== (expectedMonth + 12) % 12) {
                d.setDate(0); 
              }
              rows.push({ name, email, taskDescription: desc, frequency, dueDate: d.toISOString().split('T')[0], proofRequired: proof, verifiedBy });
            }
          } 
          else if (frequency === 'Quarterly') {
            // Generate 4 quarters (1 complete year)
            for (let i = 0; i < 4; i++) {
              let d = new Date(startDate);
              let expectedMonth = (d.getMonth() + (i * 3)) % 12;
              d.setMonth(d.getMonth() + (i * 3));
              
              if (d.getMonth() !== expectedMonth && d.getMonth() !== (expectedMonth + 12) % 12) {
                d.setDate(0);
              }
              rows.push({ name, email, taskDescription: desc, frequency, dueDate: d.toISOString().split('T')[0], proofRequired: proof, verifiedBy });
            }
          } 
          else {
            // If frequency is 'Yearly' or none, just use the dates manually clicked
            dates.forEach(dueDate => rows.push({ name, email, taskDescription: desc, frequency, dueDate, proofRequired: proof, verifiedBy }));
          }
        });
        return rows;
      }

      function submitBulk() {
        const rows = collectBulkRows();
        if (!rows.length) { showToast('Add at least one complete task with due date(s)','warning'); return; }
        const btn = document.getElementById('submitBulkBtn');
        btn.disabled = true; btn.innerHTML = '<i class="fas fa-spinner fa-pulse mr-2"></i> Saving…';
        google.script.run

            showToast(err.message,'error');
          })
          .bulkInsertChecklist(rows);
      }

      let calCurrentMonth = new Date().getMonth();
      let calCurrentYear = new Date().getFullYear();

      function openCalendar(tr, triggerEl, initialDates, frequency, onApplyCallback) {
        document.querySelectorAll('.bulk-date-trigger').forEach(el => el.classList.remove('active-trigger'));
        
        calActiveTr = tr;
        calActiveTrigger = triggerEl;
        calActiveTrigger.classList.add('active-trigger');
        calTempSelectedDates = [...initialDates];
        calFrequency = frequency || '';
        calOnApplyCallback = onApplyCallback;

        if (calTempSelectedDates.length > 0) {
          const firstDate = new Date(calTempSelectedDates[0]);
          if (!isNaN(firstDate.getTime())) {
            calCurrentMonth = firstDate.getMonth();
            calCurrentYear = firstDate.getFullYear();
          }
        } else {
          const today = new Date();
          calCurrentMonth = today.getMonth();
          calCurrentYear = today.getFullYear();
        }

        renderCalendar();
        
        const container = document.getElementById('multiDatePickerContainer');
        if (container) {
          container.classList.remove('hidden');
        }
      }

      function closeCalendar() {
        const container = document.getElementById('multiDatePickerContainer');
        if (container) {
          container.classList.add('hidden');
        }
        if (calActiveTrigger) {
          calActiveTrigger.classList.remove('active-trigger');
        }
        calActiveTr = null;
        calActiveTrigger = null;
        calOnApplyCallback = null;
      }

      function renderCalendar() {
        const container = document.getElementById('multiDatePickerContainer');
        if (!container) return;

        const titleEl = document.getElementById('calMonthYearTitle');
        const gridEl = document.getElementById('calDaysGrid');
        const summaryEl = document.getElementById('calSelectionSummary');

        const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
        titleEl.textContent = `${months[calCurrentMonth]} ${calCurrentYear}`;

        gridEl.innerHTML = '';
        
        const firstDayIndex = new Date(calCurrentYear, calCurrentMonth, 1).getDay();
        const lastDay = new Date(calCurrentYear, calCurrentMonth + 1, 0).getDate();
        const prevLastDay = new Date(calCurrentYear, calCurrentMonth, 0).getDate();

        for (let i = firstDayIndex; i > 0; i--) {
          const dayDiv = document.createElement('div');
          dayDiv.className = 'p-1.5 text-xs text-gray-300 select-none';
          dayDiv.textContent = prevLastDay - i + 1;
          gridEl.appendChild(dayDiv);
        }

        const todayStr = getMinDate();
        for (let day = 1; day <= lastDay; day++) {
          const monthStr = String(calCurrentMonth + 1).padStart(2, '0');
          const dayStr = String(day).padStart(2, '0');
          const dateStr = `${calCurrentYear}-${monthStr}-${dayStr}`;
          
          const isPast = dateStr < todayStr;
          const isSelected = calTempSelectedDates.includes(dateStr);

          const dayBtn = document.createElement('button');
          dayBtn.type = 'button';
          dayBtn.className = 'w-8 h-8 rounded-full text-xs font-semibold flex items-center justify-center transition-all duration-100 ';
          dayBtn.textContent = day;

          if (isPast) {
            dayBtn.className += 'text-gray-300 cursor-not-allowed';
            dayBtn.disabled = true;
          } else {
            if (isSelected) {
              dayBtn.className += 'bg-blue-600 text-white shadow-sm hover:bg-blue-700';
            } else {
              dayBtn.className += 'text-gray-700 hover:bg-blue-50';
            }

            dayBtn.addEventListener('click', (e) => {
              e.preventDefault();
              e.stopPropagation(); // Stop click from bubbling up and auto-closing calendar
              toggleDateSelection(dateStr);
            });
          }
          gridEl.appendChild(dayBtn);
        }

        summaryEl.textContent = `${calTempSelectedDates.length} selected`;
      }

      function toggleDateSelection(dateStr) {
        if (calFrequency === 'Daily') {
          calTempSelectedDates = [dateStr];
          applyCalendarSelection(); // Auto apply and close for Daily frequency!
        } else {
          if (calTempSelectedDates.includes(dateStr)) {
            calTempSelectedDates = calTempSelectedDates.filter(d => d !== dateStr);
          } else {
            calTempSelectedDates.push(dateStr);
          }
          renderCalendar();
        }
      }

      function applyCalendarSelection() {
        if (calOnApplyCallback) {
          calOnApplyCallback(calTempSelectedDates);
        }
        closeCalendar();
      }
