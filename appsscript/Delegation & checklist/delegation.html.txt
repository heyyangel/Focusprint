  function parseDateTimeString(str) {
    if (!str) return null;
    let d = new Date(str);
    if (!isNaN(d.getTime())) return d;
    
    // Manual parse for "dd mmm yy HH:mm:ss" format
    try {
      const parts = String(str).trim().split(/\s+/);
      if (parts.length >= 4) {
        const day = parseInt(parts[0]);
        const monthStr = parts[1];
        const yearStr = parts[2];
        const timeStr = parts[3];
        
        const monthNames = {
          jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5,
          jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11
        };
        const month = monthNames[monthStr.toLowerCase().substring(0, 3)];
        if (month === undefined) return null;
        let year = parseInt(yearStr);
        if (year < 100) year += 2000;
        
        const timeParts = timeStr.split(':');
        const hrs = parseInt(timeParts[0] || 0);
        const mins = parseInt(timeParts[1] || 0);
        const secs = parseInt(timeParts[2] || 0);
        
        return new Date(year, month, day, hrs, mins, secs);
      }
    } catch (e) {
      return null;
    }
    return null;
  }

    // Date safety check

    // 2. Manual Date Range Filter

        let match = true;
        if (fromDate) match = match && (dateStr >= fromDate);
        if (toDate) match = match && (dateStr <= toDate);
        return match;
      });
    } 
  

    // 3. Search Query Logic
    if (searchQuery) {
      const q = searchQuery.toLowerCase();

function applyRoleFilter(data) {
    const normUser = (currentUser || '').trim().toLowerCase();
    
    return data.filter(r => {
      const isDelegator = (r['Delegator'] || '').trim().toLowerCase() === normUser;
      const isDelegatee = (r['Delegetee'] || '').trim().toLowerCase() === normUser;
      const isSubMonitor = (r['Sub Monitoring'] || '').trim().toLowerCase() === normUser;
      const isCompleted = String(r['Status'] || '').includes('Completed');
      
      // AUTO-REMOVE FEATURE for Sub-Monitors
      if (isSubMonitor && !isDelegator && !isDelegatee && isCompleted) {
        return false;
      }
      
      // LOGIC FIX for All / By Me / To Me tabs
      if (currentRoleSwitcher === 'all') return true; // Show everything if 'All' is selected
      if (currentRoleSwitcher === 'byme') return isDelegator || isSubMonitor;
      if (currentRoleSwitcher === 'tome') return isDelegatee || isSubMonitor;
      
      // Default: If role is Delegatee
      if (currentRole === 'Delegatee') return isDelegatee || isSubMonitor;
      
      return false; 
    });
}

    function updateRowHighlight(tr, checked) {

    function updateBulkDeleteBtnState() {

      const bulkContainer = document.getElementById('bulkDeleteBtnContainer');
      const countLabels = bulkContainer ? bulkContainer.querySelectorAll('.selected-count') : [];
      if (bulkContainer) {
        if (checkedCbs.length > 0) {
          countLabels.forEach(el => el.textContent = checkedCbs.length);
          bulkContainer.classList.remove('translate-y-20', 'opacity-0', 'pointer-events-none');
          bulkContainer.classList.add('translate-y-0', 'opacity-100');
        } else {
          bulkContainer.classList.add('translate-y-20', 'opacity-0', 'pointer-events-none');
          bulkContainer.classList.remove('translate-y-0', 'opacity-100');
        }
      }
    }

  // ── Pagination ─────────────────────────────────────────────────────────
  function renderPagination(total, totalPages) {
    const pgBar = document.getElementById('paginationBar');
    if (totalPages <= 1) { pgBar.classList.add('hidden'); return; }
    pgBar.classList.remove('hidden');
    const start = (currentPage-1)*pageSize+1, end = Math.min(currentPage*pageSize, total);
    document.getElementById('paginationInfo').textContent = `Showing ${start}–${end} of ${total} tasks`;
    const pgNumbers = document.getElementById('pgNumbers');
    pgNumbers.innerHTML = '';
    let sp = Math.max(1, currentPage-2), ep = Math.min(totalPages, sp+4);
    if (ep-sp < 4) sp = Math.max(1, ep-4);
    for (let p = sp; p <= ep; p++) {
      const btn = document.createElement('button');
      btn.textContent = p;
      btn.className = `w-7 h-7 rounded-lg text-xs font-semibold border transition ${p===currentPage?'bg-violet-600 text-white border-violet-600 shadow':'bg-white text-gray-600 border-gray-200 hover:bg-violet-50 hover:border-violet-300'}`;
      btn.addEventListener('click', () => { currentPage=p; renderTable(filteredCache); });
      pgNumbers.appendChild(btn);
    }
    const pf=document.getElementById('pgFirst'),pp=document.getElementById('pgPrev'),pn=document.getElementById('pgNext'),pl=document.getElementById('pgLast');
    pf.disabled=pp.disabled=currentPage===1;
    pn.disabled=pl.disabled=currentPage===totalPages;
    pf.onclick=()=>{currentPage=1;renderTable(filteredCache);};
    pp.onclick=()=>{if(currentPage>1){currentPage--;renderTable(filteredCache);}};
    pn.onclick=()=>{if(currentPage<totalPages){currentPage++;renderTable(filteredCache);}};
    pl.onclick=()=>{currentPage=totalPages;renderTable(filteredCache);};
  }

function openAddModal() {
  editingTaskId = null;
  document.getElementById('addModalTitle').innerHTML = '<i class="fas fa-plus-circle text-violet-500 mr-2"></i> Add Task';
  document.getElementById('submitAddBtn').innerHTML = '<i class="fas fa-save mr-2"></i> Save Task';
  clearForm();

  // Populate dropdowns fresh each time modal opens
  populateDelegatorOptions();

  const fDelegator = document.getElementById('f_delegator');
  const fDelegatee = document.getElementById('f_delegatee');
  const normRole = String(currentRole || '').trim().toLowerCase();
  const isLockedRole = (normRole === 'delegator' || normRole === 'delegatee' || normRole === 'user');

  if (fDelegator) {
    // Always ensure the logged-in user exists as an option
    if (currentUser) {
      const exists = [...fDelegator.options].some(o => o.value === currentUser);
      if (!exists) {
        const opt = document.createElement('option');
        opt.value = currentUser;
        opt.textContent = currentUser;
        fDelegator.appendChild(opt);
      }
      // Default the field to the logged-in user for EVERY role
      fDelegator.value = currentUser;
    }

    if (isLockedRole) {
      fDelegator.setAttribute('disabled', 'true');
      fDelegator.style.backgroundColor = '#f3f4f6';
      fDelegator.style.cursor = 'not-allowed';
    } else {
      // Admin / PC / Boss — pre-filled but still changeable
      fDelegator.removeAttribute('disabled');
      fDelegator.style.backgroundColor = '';
      fDelegator.style.cursor = '';
    }
  }

  if (fDelegatee) {
    fDelegatee.removeAttribute('disabled');
    fDelegatee.style.backgroundColor = '';
    fDelegatee.style.cursor = '';
  }

  // Refresh delegatee list based on the now-set delegator
  updateDelegateeOptions();
  initSubMonitoringVisibility();
  openModal('addModal', 'addSlidePanel');
}

} else {

}
    // Due Date parsing for datetime-local

  function exportToCSV() {
    if (!delegationData.length) { showToast('No data to export','warning'); return; }
    const cols = ['Timestamp','Task Id','Delegator','Delegetee','Parent Task','Department','Task Title','Task','Prority','Score','Due Date','Status','Proof Required','Approval Required','Document Required','Revised Date','Count of Revisions','Remark','Proof of Completion','Completion Time'];
    const rows = [cols.join(',')];
    delegationData.forEach(r => {
      rows.push(cols.map(c => {
        const v = r[c]!=null ? String(r[c]) : '';
        return v.includes(',')||v.includes('"')||v.includes('\n') ? `"${v.replace(/"/g,'""')}"` : v;
      }).join(','));
    });
    const blob = new Blob([rows.join('\n')],{type:'text/csv'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href=url; a.download=`delegation_export_${new Date().toISOString().slice(0,10)}.csv`;
    a.click(); URL.revokeObjectURL(url);
    showToast('Exported successfully');
  }

          unauthorizedTasks.push(`${row['Task Id']} (${row['Task Title']})`);
        }
      }
    });

        icon: 'error',
        title: 'Permission Denied',
        text: 'Only the Boss or the Task Delegator can add remarks or update status for these tasks.',
        confirmButtonColor: '#6366f1'
      });
      return;
    }

    if (unauthorizedTasks.length > 0) {
      Swal.fire({
        icon: 'info',
        title: 'Permission Notice',
        text: `Some selected tasks were skipped because you are not their Delegator: ${unauthorizedTasks.join(', ')}. Only tasks you delegated will be updated.`,

        showCancelButton: true

          document.getElementById('remarkModal').classList.remove('hidden');

            document.getElementById('remarkSlidePanel').style.transform = 'translateX(0)';

  function submitBulkRemark() {
    const rows = document.querySelectorAll('.task-bulk-row');
    const updates = [];
    let hasEmptyRemark = false;

    rows.forEach(r => {
      const taskId = r.getAttribute('data-task-id');
      const status = r.querySelector('.bulk-row-status').value;
      const remarkInput = r.querySelector('.bulk-row-remark');
      const remark = remarkInput.value.trim();

      if (!remark) {
        hasEmptyRemark = true;
        remarkInput.classList.add('border-red-500', 'focus:ring-red-200', 'bg-red-50/50');
      } else {
        remarkInput.classList.remove('border-red-500', 'focus:ring-red-200', 'bg-red-50/50');
      }

      updates.push({ taskId, status, remark, user: currentUser });
    });

    if (hasEmptyRemark) {

  function viewRemarkHistory(taskId) {
    const row = delegationData.find(r => String(r['Task Id']) === String(taskId));
    if (!row) return;

    let history = [];
    const rawRemark = row['Remark'];
    if (rawRemark) {
      try {
        history = JSON.parse(rawRemark);
        if (!Array.isArray(history)) {
          history = [{ time: 'Original', remark: String(rawRemark), status: '' }];
        }
      } catch (e) {
        history = [{ time: 'Original', remark: String(rawRemark), status: '' }];
      }
    }

    if (history.length === 0) {
      Swal.fire({
        title: 'No History',
        text: 'No remarks have been recorded for this task yet.',
        icon: 'info',
        confirmButtonColor: '#6366f1'
      });
      return;
    }

    let timelineHTML = `<div class="text-left space-y-4 max-h-[350px] overflow-y-auto pr-1">`;
    history.forEach((h, index) => {
      const isLast = index === history.length - 1;
      const statusPill = h.status && h.status !== 'No Change' 
        ? `<span class="bg-violet-100 text-violet-800 text-[10px] font-bold px-1.5 py-0.5 rounded-full ml-2">${h.status}</span>` 
        : '';
      timelineHTML += `
        <div class="flex gap-3">
          <div class="flex flex-col items-center shrink-0">
            <div class="w-2.5 h-2.5 rounded-full ${isLast ? 'bg-violet-600 ring-4 ring-violet-100' : 'bg-gray-300'}"></div>
            ${!isLast ? '<div class="w-0.5 flex-1 bg-gray-200 my-1"></div>' : ''}
          </div>
          <div class="bg-gray-50 border border-gray-100 rounded-xl p-3 flex-1 text-xs text-gray-700">
            <div class="flex justify-between items-center mb-1.5 gap-2">
              <span class="font-mono text-gray-400 text-[10px]">${h.time || ''}</span>
              <div class="flex items-center gap-1.5 shrink-0">
                ${h.user ? `<span class="bg-gray-200 text-gray-600 text-[9px] font-medium px-1.5 py-0.5 rounded"><i class="fas fa-user text-[8px] mr-1 text-gray-400"></i>${esc(h.user)}</span>` : ''}
                ${statusPill}
              </div>
            </div>
            <div class="font-semibold text-gray-800 break-words leading-relaxed">${esc(h.remark)}</div>
          </div>
        </div>`;
    });
    timelineHTML += `</div>`;

    Swal.fire({
      title: `<span class="text-lg font-bold text-gray-800 flex items-center gap-2"><i class="fas fa-history text-violet-500"></i> Remark Timeline</span>`,
      html: timelineHTML,
      showConfirmButton: true,
      confirmButtonText: 'Close',
      confirmButtonColor: '#6366f1',
      customClass: {
        popup: 'rounded-2xl shadow-xl'
      }
    });
  }

      if (matchedUser && matchedUser.department) {
        const deptDropdown = document.getElementById('f_department');
        if (deptDropdown) {
          // Check if the department is already in the dropdown options
          const optionExists = [...deptDropdown.options].some(opt => opt.value === matchedUser.department);
          
          // If it's a new department not currently in the dropdown, add it dynamically
          if (!optionExists) {
            const newOpt = document.createElement('option');
            newOpt.value = matchedUser.department;
            newOpt.textContent = matchedUser.department;
            deptDropdown.appendChild(newOpt);
          }
          
          // Auto-select the department in the dropdown
          deptDropdown.value = matchedUser.department;
        }
      }
    });
  });

   function getLocalISODate(d) {
    if (!d || isNaN(d.getTime())) return '';
    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${year}-${month}-${day}`;
  }
