        // ── Helpers ────────────────────────────────────────────────────────────
        function showToast(msg, icon = 'success') {
            Swal.fire({ toast: true, position: 'top-end', icon, title: msg, showConfirmButton: false, timer: 2200, timerProgressBar: true });
        }
        function escapeHtml(s) {
            return s ? String(s).replace(/[&<>"']/g, m => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[m])) : '';
        }
        function getLatestRemarkText(remarkValue) {
            if (!remarkValue) return '—';
            const val = String(remarkValue).trim();
            if (val.startsWith('[') && val.endsWith(']')) {
                try {
                    const arr = JSON.parse(val);
                    if (Array.isArray(arr) && arr.length > 0) {
                        const last = arr[arr.length - 1];
                        return last.remark || last.text || '—';
                    }
                } catch (e) {
                    console.error("Error parsing remark JSON", e);
                }
            }
            return val || '—';
        }

        // ── Delay formatting: show days if >= 24h, else hours ─────────────────
        function computeDelayText(dueDateStr, isCompleted) {
            if (isCompleted || !dueDateStr) return '';
            const due = new Date(dueDateStr);
            if (isNaN(due.getTime())) return '';
            const now = new Date();
            if (due >= now) return '';
            const diffMs = now - due;
            const totalMinutes = Math.floor(diffMs / (1000 * 60));
            const totalHours = Math.floor(diffMs / (1000 * 60 * 60));
            const totalDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
            if (totalDays >= 1) return ` (${totalDays} day${totalDays > 1 ? 's' : ''} delay)`;
            if (totalHours >= 1) return ` (${totalHours}h delay)`;
            if (totalMinutes >= 1) return ` (${totalMinutes}m delay)`;
            return ' (just delayed)';
        }

    : [{ label: 'By Me', val: 'byme' }, { label: 'To Me', val: 'tome' }];
currentRoleSwitcher = options[0].val;

function applyFiltersAndRender() {
    const normUser = (currentUser || '').trim().toLowerCase();

    // 1. Visibility & Role Logic (User check)
  // 1. Visibility & Role Logic (User check)
    const isVisibleForUser = (t) => {
        const isDelegator = (t['Verified by'] || '').trim().toLowerCase() === normUser;
        const isDelegatee = (t.Name || '').trim().toLowerCase() === normUser;
        const isSubMonitor = (t['Sub Monitoring'] || '').trim().toLowerCase() === normUser;
        const isCompleted = (t['Actual'] || '').trim() !== '';

        if (isSubMonitor && !isDelegator && !isDelegatee && isCompleted) {
            return false; // Auto-remove completed sub-monitored tasks
        }

        // Agar normal doer hai
        if (currentRole === 'Delegatee' || currentRole === 'User') {
            return isDelegatee || isSubMonitor || isDelegator;
        }

        // Agar "All" tab selected hai
        // Agar "All" tab selected hai — Delegator/Admin/PC sabko poora data dikhega
if (currentRoleSwitcher === 'all') {
    return true;
}

        // Agar "By Me" tab selected hai
        if (currentRoleSwitcher === 'byme') return isDelegator || isSubMonitor;
        
        // Agar "To Me" tab selected hai
        if (currentRoleSwitcher === 'tome') return isDelegatee || isSubMonitor;

        return false;
    };

    let data = [...allTasks];

    // Apply Role Filter
    data = data.filter(isVisibleForUser);

    // 2. Tab filter (Checklist vs Delegation)
    if (currentTab === 'delegation') {
        data = data.filter(t => t._type === 'delegation');
    } else if (currentTab === 'checklist') {
        data = data.filter(t => t._type === 'checklist');
    }

    const fromDate = document.getElementById('filterFromDate').value;
    const toDate = document.getElementById('filterToDate').value;

    const todayLocal = new Date();
    const todayStr = `${todayLocal.getFullYear()}-${String(todayLocal.getMonth() + 1).padStart(2, '0')}-${String(todayLocal.getDate()).padStart(2, '0')}`;

    if ((fromDate && fromDate > todayStr) || (toDate && toDate > todayStr)) {
        showToast("Future dates cannot be selected!", "error");
        return;
    }

    // 3. Date Filtering Logic
    if (fromDate || toDate) {
        data = data.filter(t => {
            const dStr = t['Revised Date'] || t['Due Date'];
            if (!dStr) return false;
            const d = parseDateTimeString(dStr);
            if (!d) return false;
            const dateStr = getLocalISODate(d);
            let match = true;
            if (fromDate) match = match && (dateStr >= fromDate);
            if (toDate) match = match && (dateStr <= toDate);
            return match;
        });
    } else if (currentQuickFilter === 'delayed') {
        data = data.filter(t => {
            if ((t['Actual'] || '').trim() !== '') return false;
            const dStr = t['Revised Date'] || t['Due Date'];
            if (!dStr) return false;
            const d = parseDateTimeString(dStr);
            return d && d < new Date();
        });
    } else if (currentQuickFilter === 'today') {
        const todayStart = new Date(); todayStart.setHours(0, 0, 0, 0);
        const todayEnd = new Date(); todayEnd.setHours(23, 59, 59, 999);
        data = data.filter(t => {
            const dStr = t['Revised Date'] || t['Due Date'];
            if (!dStr) return false;
            const d = parseDateTimeString(dStr);
            return d && d >= todayStart && d <= todayEnd;
        });
    } 
    // Upcoming filter logic
    else if (currentQuickDate === 'upcoming') {
        const today = new Date(); today.setHours(0, 0, 0, 0);
        const tomorrow = new Date(); tomorrow.setDate(today.getDate() + 1); tomorrow.setHours(23, 59, 59, 999);

        data = data.filter(t => {
            // Delegation tab ke liye saare tasks dikhao (Date bypass)
            if (t._type === 'delegation') return true;

            // Checklist tab ke liye aaj aur kal ka filter
            const dStr = t['Revised Date'] || t['Due Date'];
            if (!dStr) return false;
            const d = parseDateTimeString(dStr);
            if (!d) return false;

            const isTodayOrTomorrow = d.getTime() >= today.getTime() && d.getTime() <= tomorrow.getTime();
            const isCompleted = (t['Actual'] || '').trim() !== '';
            const isPastDelayed = d.getTime() < today.getTime() && !isCompleted;

            return isTodayOrTomorrow || isPastDelayed;
        });
    }

    // 4. Multi-Select Filters
    if (filterNames.length) data = data.filter(t => filterNames.includes(t.Name));
    if (filterFreqs.length) data = data.filter(t => filterFreqs.includes(t.Frequency));
    if (filterVerified.length) data = data.filter(t => filterVerified.includes((t['Verified by'] || '').trim()));

    // 5. Status Filtering (Pending vs Completed)
    if (filterStatus.length) {
    data = data.filter(t => {
        const isCompleted = (t['Actual'] || '').trim() !== '';
        if (filterStatus.includes('pending') && !isCompleted) return true;
        if (filterStatus.includes('completed') && isCompleted) return true;
        return false;
    });
} else {
        data = data.filter(t => {
            const isCompleted = (t['Actual'] || '').trim() !== '';
            if (t._type === 'checklist') {
                return !isCompleted; // Checklist mein sirf pending dikhao
            } else {
                return true; // Delegation mein sab dikhao
            }
        });
        data = data.filter(t => (t['Actual'] || '').trim() === ''); 
    }

    // 6. Fix Tab Counts (Counts strictly pending tasks for both tabs)
    const baseForCounts = allTasks.filter(isVisibleForUser).filter(t => (t['Actual'] || '').trim() === '');

    document.getElementById('countDelegation').textContent = baseForCounts.filter(t => t._type === 'delegation').length;
    document.getElementById('countChecklist').textContent = baseForCounts.filter(t => t._type === 'checklist').length;

    filteredTasks = data;
    updateFilterBadge();
    renderCurrentView();
}

                        if (diffDays > 0) gapHtml = `<div class="text-[9px] font-bold text-red-600 mt-0.5">+${diffDays}d late</div>`;
                        else if (diffDays < 0) gapHtml = `<div class="text-[9px] font-bold text-green-600 mt-0.5">${Math.abs(diffDays)}d early</div>`;
                        else gapHtml = `<div class="text-[9px] font-bold text-blue-600 mt-0.5">On time</div>`;
                    }
                } else if (isOverdue) {
                    const due = new Date(dueDateStr);
                    const now = new Date();
                    const diffDays = Math.floor((now - due) / (1000 * 60 * 60 * 24));
                    gapHtml = `<div class="text-[9px] font-bold text-red-600 mt-0.5">${diffDays > 0 ? diffDays + 'd overdue' : 'Delayed'}</div>`;

                if (!isCompleted) {
                  actionButtonsHtml += '<button type="button" class="w-7 h-7 flex items-center justify-center rounded-lg bg-emerald-50 text-emerald-600 hover:bg-emerald-100 border border-emerald-100 transition active:scale-95" title="Complete Task" onclick="event.stopPropagation(); triggerRowAction(\'' + escapeHtml(task._type + '_' + taskId) + '\', \'complete\')"><i class="fas fa-check text-[10px]"></i></button> ';
                  actionButtonsHtml += '<button type="button" class="w-7 h-7 flex items-center justify-center rounded-lg bg-indigo-50 text-indigo-600 hover:bg-indigo-100 border border-indigo-100 transition active:scale-95" title="Revise Due Date" onclick="event.stopPropagation(); triggerRowAction(\'' + escapeHtml(task._type + '_' + taskId) + '\', \'revise\')"><i class="fas fa-calendar-alt text-[10px]"></i></button>';

                // --- NEW TABLE VERIFICATION LOGIC ---
                const verifierName = (task['Verified by'] || '').trim();
                const isVerified = (task._type === 'checklist')
                  ? (task['Verified'] === true || String(task['Verified']).toLowerCase() === 'true')
                  : (task['Status'] === 'Completed' || task['Status'] === 'Completed With Delay');
                const verifiedTs = task['Verified Timestamp'] || '';

                let tableVerificationHTML = '';
                if (!verifierName || verifierName === '') {
                  tableVerificationHTML = '<span class="text-gray-300">—</span>';
                } else {
                  if (isVerified) {
                    tableVerificationHTML = `
                      <div class="font-bold text-emerald-600 flex items-center gap-1">
                        <i class="fas fa-check-double text-[10px]"></i> Verified by ${escapeHtml(verifierName)}
                      </div>
                      ${verifiedTs ? `<div class="text-[9px] text-gray-400">At: ${fmtDateTime(verifiedTs)}</div>` : ''}

                      <div class="font-bold text-amber-600 flex items-center gap-1" title="Pending verification by ${escapeHtml(verifierName)}">

                      </div>

                }

                return `
                    <tr class="task-row ${isSelected ? 'selected' : ''}" data-task-id="${escapeHtml(task._type + '_' + taskId)}">
                        <td><input type="checkbox" class="row-select-cb accent-indigo-600" data-task-id="${escapeHtml(task._type + '_' + taskId)}" ${isSelected ? 'checked' : ''}></td>
                        <td class="font-mono text-[10px] font-bold text-gray-400">${escapeHtml(taskId)}</td>
             <td class="font-semibold text-gray-700">
                            <div class="truncate max-w-[150px]" title="${escapeHtml(task.Name || '—')}">${escapeHtml(task.Name || '—')}</div>
                            ${(() => {
                                const subMon = (task['Sub Monitoring'] || '').trim();
                                if (subMon && subMon !== 'No') {
                                    return `<div class="mt-1 inline-flex items-center gap-1 bg-indigo-50 border border-indigo-200 text-indigo-700 px-1.5 py-0.5 rounded text-[9px] font-bold shadow-sm" title="Monitored by: ${escapeHtml(subMon)}"><i class="fas fa-eye" style="font-size:8px;"></i> Mon: ${escapeHtml(subMon)}</div>`;
                                }
                                return '';
                            })()}
                        </td>
                        <td class="min-w-[250px] max-w-[400px]">
          <div class="break-words whitespace-normal">${escapeHtml(task['Task Description'] || '—')}</div>
        </td>
                        <td><span class="px-2 py-0.5 rounded-full text-[10px] font-bold" style="${getFreqStyle(task.Frequency)}">${escapeHtml(task.Frequency || 'Once')}</span></td>

                            <div>${fmtDate(task['Due Date'])}</div>
                            ${task['Revised Date'] ? `<div class="text-[10px] text-indigo-600 font-bold mt-0.5" title="Revised Due Date"><i class="far fa-calendar-alt text-[9px] mr-0.5"></i>Rev: ${fmtDate(task['Revised Date'])}</div>` : ''}

                        <td class="whitespace-nowrap">${actual ? fmtDate(actual) : '—'}</td>

                cb.addEventListener('change', function(e) {

                    else selectedTaskIds.delete(tid);

                    if (row) row.classList.toggle('selected', this.checked);

        function attachTableRowClickHandlers() {
            document.querySelectorAll('.task-row').forEach(row => {
                row.addEventListener('click', function(e) {
                    if (e.target.closest('input') || e.target.closest('a')) return;
                    const tid = this.getAttribute('data-task-id');
                    const cb = this.querySelector('.row-select-cb');
                    if (!cb) return;
                    cb.checked = !cb.checked;
                    if (cb.checked) selectedTaskIds.add(tid);
                    else selectedTaskIds.delete(tid);
                    this.classList.toggle('selected', cb.checked);
                    updateSelectAllState();
                    updateCompleteOverlay();
                });
            });
        }

        function updateSelectAllState() {
            const cbs = document.querySelectorAll(currentView === 'card' ? '.card-select-cb' : '.row-select-cb');
            const allChecked = cbs.length > 0 && [...cbs].every(cb => cb.checked);
            const someChecked = [...cbs].some(cb => cb.checked);
            const selectAllCb = document.getElementById('selectAllCb');
            if (selectAllCb) {
                selectAllCb.checked = allChecked;
                selectAllCb.indeterminate = someChecked && !allChecked;
            }
            const count = selectedTaskIds.size;
            const countEl = document.getElementById('selectedCount');
            if (countEl) {
                if (count > 0) {
                    countEl.textContent = `${count} selected`;
                    countEl.classList.remove('hidden');
                } else {
                    countEl.classList.add('hidden');
                }
            }

            // Show/hide unselect button
            const clearBtn = document.getElementById('clearSelectionBtn');
            if (clearBtn) {
                if (count > 0) clearBtn.classList.remove('hidden'), clearBtn.classList.add('flex');
                else clearBtn.classList.add('hidden'), clearBtn.classList.remove('flex');
            }

            updateCompleteOverlay();
        }

            const cbs = document.querySelectorAll(currentView === 'card' ? '.card-select-cb' : '.row-select-cb');

        function updateCompleteOverlay() {
            const normUser = (currentUser || '').trim().toLowerCase();
            const actionOverlay = document.getElementById('actionOverlay');
            const completeBtn = document.getElementById('completeOverlayBtn');
            const verifyBtn = document.getElementById('verifyOverlayBtn');

            const count = selectedTaskIds.size;
            const selectedTasks = allTasks.filter(t => selectedTaskIds.has(t._type + '_' + t['Task ID']));

            // Tasks I can complete (must be pending)
            const completableTasks = selectedTasks.filter(t => (t['Actual'] || '').trim() === '');
            // Tasks I can verify (must be completed AND I must be the verifier AND not yet verified)
            const verifiableTasks = selectedTasks.filter(t =>
                (t['Actual'] || '').trim() !== '' &&
                (t['Verified by'] || '').trim().toLowerCase() === normUser &&
                !(t['Verified Timestamp'])
            );

            document.getElementById('overlayCount').textContent = completableTasks.length;
            document.getElementById('verifyOverlayCount').textContent = verifiableTasks.length;

            // Show/hide specific buttons
            if (completableTasks.length > 0) {
                completeBtn.classList.remove('hidden');
                completeBtn.classList.add('flex');
            } else {
                completeBtn.classList.add('hidden');
                completeBtn.classList.remove('flex');
            }

            if (verifiableTasks.length > 0) {
                verifyBtn.classList.remove('hidden');
                verifyBtn.classList.add('flex');
            } else {
                verifyBtn.classList.add('hidden');
                verifyBtn.classList.remove('flex');
            }

            // Show/hide entire overlay
            if (completableTasks.length > 0 || verifiableTasks.length > 0) {
                actionOverlay.classList.add('visible');
            } else {
                actionOverlay.classList.remove('visible');
            }
        }

        async function submitVerify() {
    const btn = document.getElementById('verifyOverlayBtn');
    const selectedTasks = allTasks.filter(t => selectedTaskIds.has(t._type + '_' + t['Task ID']));

    const verifiableTaskIds = selectedTasks
        .filter(t => (t['Actual'] || '').trim() !== '' && !t['Verified Timestamp'])
        .map(t => t['Task ID']);

    if (!verifiableTaskIds.length) {
        showToast('No tasks eligible for verification', 'info');
        return;

    // Confirmation pop-up
    const confirm = await Swal.fire({

        text: `Mark ${verifiableTaskIds.length} task(s) as verified?`,
        icon: 'question',
        showCancelButton: true,
        confirmButtonText: 'Yes, Verify'
    });
    if (!confirm.isConfirmed) return;

    btn.disabled = true;
    btn.innerHTML = '<i class="fas fa-spinner fa-spin mr-2"></i>Verifying…';

    // FIX: Yahan 'currentUser' ka naam pass karein
    google.script.run

        function viewRemarkHistory(compositeId) {
            const task = allTasks.find(t => (t._type + '_' + t['Task ID']) === compositeId);
            if (!task) return;

            let history = [];
            const rawRemark = task['Remark'];
            if (rawRemark) {
                try {
                    history = JSON.parse(rawRemark);
                    if (!Array.isArray(history)) {
                        history = [{ time: 'Original', remark: String(rawRemark), status: '' }];
                    }
                } catch (e) {
                    history = [{ time: 'Original', remark: String(rawRemark), status: '' }];
                }
            }

            if (history.length === 0) {
                Swal.fire({
                    title: 'No History',
                    text: 'No remarks have been recorded for this task yet.',
                    icon: 'info',
                    confirmButtonColor: '#6366f1'
                });
                return;
            }

            let timelineHTML = `<div class="text-left space-y-4 max-h-[350px] overflow-y-auto pr-1">`;
            history.forEach((h, index) => {
                const isLast = index === history.length - 1;
                const statusPill = h.status && h.status !== 'No Change' 
                    ? `<span class="bg-indigo-100 text-indigo-800 text-[10px] font-bold px-1.5 py-0.5 rounded-full ml-2">${h.status}</span>` 
                    : '';
                timelineHTML += `
                    <div class="flex gap-3" style="margin-bottom: 12px;">
                        <div class="flex flex-col items-center shrink-0">
                            <div class="w-2.5 h-2.5 rounded-full ${isLast ? 'bg-indigo-600 ring-4 ring-indigo-100' : 'bg-gray-300'}" style="width: 10px; height: 10px; border-radius: 50%;"></div>
                            ${!isLast ? '<div class="w-0.5 flex-1 bg-gray-200 my-1" style="width: 2px; flex: 1; background: #e2e8f0; margin-top: 4px; margin-bottom: 4px;"></div>' : ''}
                        </div>
                        <div class="bg-gray-50 border border-gray-100 rounded-xl p-3 flex-1 text-xs text-gray-700" style="background: #f8fafc; border: 1px solid #e2e8f0; border-radius: 12px; padding: 12px; flex: 1;">
                            <div class="flex justify-between items-center mb-1.5 gap-2" style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 6px;">
                                <span class="font-mono text-gray-400 text-[10px]" style="color: #94a3b8; font-family: monospace; font-size: 10px;">${h.time || ''}</span>
                                <div class="flex items-center gap-1.5 shrink-0" style="display: flex; align-items: center; gap: 6px;">
                                    ${h.user ? `<span class="bg-gray-200 text-gray-600 text-[9px] font-medium px-1.5 py-0.5 rounded" style="background: #e2e8f0; color: #475569; padding: 2px 6px; border-radius: 4px; font-size: 9px;"><i class="fas fa-user text-[8px] mr-1 text-gray-400" style="font-size: 8px; margin-right: 4px; color: #94a3b8;"></i>${escapeHtml(h.user)}</span>` : ''}
                                    ${statusPill}
                                </div>
                            </div>
                            <div class="font-semibold text-gray-800 break-words leading-relaxed" style="font-weight: 600; color: #1e293b; word-break: break-word; line-height: 1.5; font-size: 12px;">${escapeHtml(h.remark)}</div>
                        </div>
                    </div>`;
            });
            timelineHTML += `</div>`;

            Swal.fire({
                title: `<span class="text-lg font-bold text-gray-800 flex items-center gap-2" style="font-size: 18px; font-weight: 700; color: #1e293b; display: flex; align-items: center; gap: 8px;"><i class="fas fa-history text-indigo-500"></i> Remark Timeline</span>`,
                html: timelineHTML,
                showConfirmButton: true,
                confirmButtonText: 'Close',
                confirmButtonColor: '#4f46e5',
                customClass: {
                    popup: 'rounded-2xl shadow-xl'
                }

                }
            });

async function submitComplete() {
    const btn = document.getElementById('submitCompleteBtn');
    const selectedTasks = allTasks.filter(t => selectedTaskIds.has(t._type + '_' + t['Task ID']));
    if (!selectedTasks.length) return;

    btn.disabled = true;
    btn.innerHTML = '<i class="fas fa-spinner fa-spin mr-2"></i>Submitting…';

    let timestamp = '';
    let dateOnly = '';
    try {
        const dt = new Date();
        const day = String(dt.getDate()).padStart(2, '0');
        const mon = dt.toLocaleString('en-IN', { month: 'short' });
        const yr  = String(dt.getFullYear()).slice(-2);
        const hh  = String(dt.getHours()).padStart(2, '0');
        const mm  = String(dt.getMinutes()).padStart(2, '0');
        timestamp = `${day} ${mon} ${yr} ${hh}:${mm}:00`;
        dateOnly  = `${day} ${mon} ${yr}`;
    } catch (e) {
        timestamp = new Date().toLocaleString('en-IN', { timeZone: 'Asia/Kolkata' });
        dateOnly  = new Date().toLocaleDateString('en-GB', { day: 'numeric', month: 'short', year: '2-digit' });
    }

    try {
        // Group updates
        const completeUpdates = [];
        const reviseUpdates = [];

        for (const task of selectedTasks) {
            const taskId = task['Task ID'];

            // Dynamically read action mode for both Delegation and Checklist
            let mode = 'complete';
            const item = document.getElementById('item_' + taskId);
            if (item) {
                mode = item.getAttribute('data-action-mode') || 'complete';
            }

            if (mode === 'complete') {
                // Validate proof
                if (task['Proof is Required'] === 'Yes' && !uploadedFiles[taskId]) {
                    showToast(`Please upload proof for task: ${taskId}`, 'warning');
                    btn.disabled = false;
                    btn.innerHTML = '<i class="fas fa-check-circle mr-1.5"></i> Submit Actions';
                    return;
                }

                let documentLink = '';
                if (uploadedFiles[taskId]) {
                    const groupId = task['Group ID'] || '';
                    documentLink = await uploadFileToServer(uploadedFiles[taskId], taskId, groupId);
                }

                const remarkVal = document.getElementById('remark_complete_' + taskId)?.value.trim() || '';

                completeUpdates.push({
                    taskId,
                    documentLink,
                    actualTimestamp: (task._type === 'checklist') ? dateOnly : timestamp,
                    remark: remarkVal,
                    user: currentUser,
                    _type: task._type
                });
            } else {
                // Revise mode
                const revisedDateVal = document.getElementById('date_revise_' + taskId)?.value;
                const remarkVal = document.getElementById('remark_revise_' + taskId)?.value.trim();

                if (!revisedDateVal) {
                    showToast(`Please select a Revised Date & Time for task: ${taskId}`, 'warning');
                    btn.disabled = false;
                    btn.innerHTML = '<i class="fas fa-check-circle mr-1.5"></i> Submit Actions';
                    return;
                }
                if (!remarkVal) {
                    showToast(`Please specify a Revision Reason for task: ${taskId}`, 'warning');
                    btn.disabled = false;
                    btn.innerHTML = '<i class="fas fa-check-circle mr-1.5"></i> Submit Actions';
                    return;
                }

                let formattedRevisedDate = revisedDateVal;
                try {
                    const dt = new Date(revisedDateVal);
                    if (!isNaN(dt.getTime())) {
                        const day = String(dt.getDate()).padStart(2, '0');
                        const mon = dt.toLocaleString('en-IN', { month: 'short' });
                        const yr  = String(dt.getFullYear()).slice(-2);
                        const hh  = String(dt.getHours()).padStart(2, '0');
                        const mm  = String(dt.getMinutes()).padStart(2, '0');
                        formattedRevisedDate = `${day} ${mon} ${yr} ${hh}:${mm}:00`;
                    }
                } catch(e) {}

                reviseUpdates.push({
                    taskId,
                    revisedDate: formattedRevisedDate,
                    remark: remarkVal,
                    user: currentUser,
                    timestamp: timestamp,
                    _type: task._type // Identify category during API routing
                });
            }
        }

        // Route updates dynamically to correct Apps Script functions
        const promises = [];
        const delegationCompletes = completeUpdates.filter(u => u._type === 'delegation');
        const checklistCompletes  = completeUpdates.filter(u => u._type === 'checklist');
        const delegationRevises   = reviseUpdates.filter(u => u._type === 'delegation');
        const checklistRevises    = reviseUpdates.filter(u => u._type === 'checklist');

        if (delegationCompletes.length > 0) {
            promises.push(new Promise((resolve, reject) => {
                google.script.run
                    .withSuccessHandler(res => resolve(res))
                    .withFailureHandler(err => reject(err))
                    .completeDelegationTasks(delegationCompletes);
            }));
        }
        if (checklistCompletes.length > 0) {
            promises.push(new Promise((resolve, reject) => {
                google.script.run
                    .withSuccessHandler(res => resolve(res))
                    .withFailureHandler(err => reject(err))
                    .completeTasks(checklistCompletes);
            }));
        }
        if (delegationRevises.length > 0) {
            promises.push(new Promise((resolve, reject) => {
                google.script.run
                    .withSuccessHandler(res => resolve(res))
                    .withFailureHandler(err => reject(err))
                    .reviseDelegationTasks(delegationRevises);
            }));
        }
        if (checklistRevises.length > 0) {
            promises.push(new Promise((resolve, reject) => {
                google.script.run
                    .withSuccessHandler(res => resolve(res))
                    .withFailureHandler(err => reject(err))
                    .reviseChecklistTasks(checklistRevises);
            }));
        }

        if (promises.length === 0) {
            btn.disabled = false;
            btn.innerHTML = '<i class="fas fa-check-circle mr-1.5"></i> Submit Actions';
            return;
        }

        const results = await Promise.all(promises);

        btn.disabled = false;
        btn.innerHTML = '<i class="fas fa-check-circle mr-1.5"></i> Submit Actions';

        const failed = results.find(r => !r.success);
        if (failed) {
            showToast('Submission failed: ' + failed.message, 'error');
        } else {
            showToast('All actions submitted successfully!');
            closeCompleteSidebar();
            selectedTaskIds.clear();
            uploadedFiles = {};
            updateCompleteOverlay();
            loadTasks(true); // Silent refresh
        }

    } catch (err) {
        btn.disabled = false;
        btn.innerHTML = '<i class="fas fa-check-circle mr-1.5"></i> Submit Actions';
        showToast('Error: ' + err.message, 'error');
    }
}       
 function uploadFileToServer(file, taskId, groupId) {
            return new Promise((resolve, reject) => {
                const reader = new FileReader();
                reader.onload = function (e) {
                    const base64 = e.target.result.split(',')[1];
                    const mimeType = file.type || 'application/octet-stream';
                    // We create a blob descriptor to pass to Apps Script
                    // Google Apps Script expects a BlobSource; we pass name, data, mime
                    google.script.run
                        .withSuccessHandler(result => {
                            if (result.success) resolve(result.fileUrl);
                            else reject(new Error(result.message));
                        })
                        .withFailureHandler(err => reject(err))
                        .uploadTaskDocument(
                            // Can't pass File directly; use base64 approach via a helper
                            // Apps Script side receives this and reconstructs the blob
                            { name: file.name, data: base64, mimeType },
                            taskId, groupId, PROOF_FOLDER_ID
                        );
                };
                reader.onerror = () => reject(new Error('File read failed'));
                reader.readAsDataURL(file);
            });
        }

        function exportToCSV() {
            if (!filteredTasks.length) { showToast('No data to export', 'warning'); return; }

                    v = v.replace(/"/g, '""');
                    if (v.includes(',') || v.includes('"') || v.includes('\n')) v = `"${v}"`;
                    return v;
                }).join(','));

            const blob = new Blob([rows.join('\n')], { type: 'text/csv' });

            const a = document.createElement('a');

        function attachCardClickHandlers() {
            document.querySelectorAll('.task-card').forEach(card => {
                card.style.cursor = 'pointer';
                card.addEventListener('click', function (e) {
                    // Don't double-fire if they clicked the checkbox directly
                    if (e.target.classList.contains('card-select-cb')) return;
                    const tid = this.getAttribute('data-task-id');
                    if (!tid) return;
                    const cb = this.querySelector('.card-select-cb');
                    if (!cb) return;
                    cb.checked = !cb.checked;
                    if (cb.checked) selectedTaskIds.add(tid);
                    else selectedTaskIds.delete(tid);
                    this.classList.toggle('selected', cb.checked);
                    this.classList.toggle('border-indigo-400', cb.checked);
                    this.classList.toggle('border-gray-100', !cb.checked);
                    updateSelectAllState();
                    updateCompleteOverlay();
                });
            });
        }

          function parseDateTimeString(str) {
    if (!str) return null;
    let d = new Date(str);
    if (!isNaN(d.getTime())) return d;
    
    try {
      const parts = String(str).trim().split(/\s+/);
      if (parts.length >= 3) {
        const day = parseInt(parts[0]);
        const monthStr = parts[1];
        const yearStr = parts[2];
        
        const monthNames = {
          jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5,
          jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11
        };
        const month = monthNames[monthStr.toLowerCase().substring(0, 3)];
        if (month === undefined) return null;
        let year = parseInt(yearStr);
        if (year < 100) year += 2000;
        
        let hrs = 0, mins = 0, secs = 0;
        if (parts[3]) {
          const timeParts = parts[3].split(':');
          hrs = parseInt(timeParts[0] || 0);
          mins = parseInt(timeParts[1] || 0);
          secs = parseInt(timeParts[2] || 0);
        }
        
        return new Date(year, month, day, hrs, mins, secs);
      }
    } catch (e) {
      return null;
    }
    return null;
  }

  function getLocalISODate(d) {
    if (!d || isNaN(d.getTime())) return '';
    const year = d.getFullYear();
    const month = String(d.getMonth() + 1).padStart(2, '0');
    const day = String(d.getDate()).padStart(2, '0');
    return `${year}-${month}-${day}`;
  }
