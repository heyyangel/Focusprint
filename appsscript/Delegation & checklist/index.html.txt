        let currentUser = null;
        let currentRole = null;

        function showLogin() {
          document.getElementById('loginContainer').classList.remove('hidden');
          document.getElementById('appContainer').classList.add('hidden');
          document.getElementById('loginPassword').value = '';
          document.getElementById('loginError').classList.add('hidden');
        }

        function showApp(user, role) {
          currentUser = user;
          currentRole = role;
          document.getElementById('loginContainer').classList.add('hidden');
          document.getElementById('appContainer').classList.remove('hidden');
          document.getElementById('profileUserName').innerText = user;
          document.getElementById('profileUserRole').innerText = role;
          localStorage.setItem('erp_logged_user', user);
          localStorage.setItem('erp_logged_role', role);
          loadPageContent('delegation');
        }

        function loadPageContent(pageType) {
          const contentDiv = document.getElementById('dynamicContent');
          contentDiv.innerHTML = `<div class="flex justify-center items-center py-16"><i class="fas fa-circle-notch fa-spin text-3xl text-blue-500"></i><span class="ml-3 text-gray-600">Loading ${pageType}...</span></div>`;
          
          google.script.run
            .withSuccessHandler(function(html) {
              insertHtmlWithScripts(html, contentDiv);
              setActiveNav(`nav${pageType.charAt(0).toUpperCase() + pageType.slice(1)}`);
            })
            .withFailureHandler(function(error) {
              contentDiv.innerHTML = `<div class="bg-red-50 border border-red-200 text-red-700 p-6 rounded-xl"><i class="fas fa-exclamation-triangle"></i> Failed to load ${pageType}: ${error.message}</div>`;
            })
            .getHtmlContent(pageType);
        }

        function insertHtmlWithScripts(htmlString, targetElement) {
          const temp = document.createElement('div');
          temp.innerHTML = htmlString;
          
          const scripts = temp.querySelectorAll('script');
          const scriptContents = [];
          scripts.forEach(script => {
            if (script.src) {
              const newScript = document.createElement('script');
              newScript.src = script.src;
              document.head.appendChild(newScript);
            } else {
              scriptContents.push(script.innerHTML);
            }
            script.remove();
          });
          
          targetElement.innerHTML = temp.innerHTML;
          scriptContents.forEach(content => {
            try {
              eval(content);
            } catch (err) {
              console.error('Script execution error:', err);
            }
          });
        }

        function setActiveNav(activeId) {
          const btns = ['navDelegation', 'navChecklist', 'navTasks'];
          btns.forEach(id => {
            const btn = document.getElementById(id);
            if (!btn) return;
            if (id === activeId) {
              btn.classList.add('bg-white', 'text-blue-600', 'shadow-sm');
              btn.classList.remove('text-gray-700');
            } else {
              btn.classList.remove('bg-white', 'text-blue-600', 'shadow-sm');
              btn.classList.add('text-gray-700');
            }
          });
        }
              
        function loadUsernameDropdown() {
          google.script.run.withSuccessHandler(function(usernames) {
            const datalist = document.getElementById('usernameOptions');
            datalist.innerHTML = '';
            usernames.forEach(uname => {
              const option = document.createElement('option');
              option.value = uname;
              datalist.appendChild(option);
            });
          }).getUsernames();
        }

        function attemptLogin() {
          const username = document.getElementById('loginUsername').value.trim();
          const password = document.getElementById('loginPassword').value;
          const errorDiv = document.getElementById('loginError');
          if (!username || !password) {
            errorDiv.innerText = 'Please enter both username and password';
            errorDiv.classList.remove('hidden');
            return;
          }
          errorDiv.classList.add('hidden');
          const loginBtn = document.getElementById('loginBtn');
          loginBtn.disabled = true;
          loginBtn.innerHTML = '<i class="fas fa-spinner fa-pulse"></i> Authenticating...';
          google.script.run.withSuccessHandler(function(result) {
            loginBtn.disabled = false;
            loginBtn.innerHTML = '<i class="fas fa-arrow-right-to-bracket"></i> Sign In';
            if (result.success) {
              localStorage.setItem('erp_session_token', result.token);
              localStorage.setItem('erp_user_email', result.email || '');
              showApp(result.username, result.role, result.department); // PASS DEPT HERE
            } else {
              errorDiv.innerText = result.message || 'Invalid username or password';
              errorDiv.classList.remove('hidden');
            }
          }).withFailureHandler(function(error) {
            loginBtn.disabled = false;
            loginBtn.innerHTML = '<i class="fas fa-arrow-right-to-bracket"></i> Sign In';
            errorDiv.innerText = 'Server error: ' + error.message;
            errorDiv.classList.remove('hidden');
          }).authenticateUserWithToken(username, password);
        }
              
        function logout() {
          localStorage.removeItem('erp_logged_user');
          localStorage.removeItem('erp_logged_role');
          localStorage.removeItem('erp_session_token');
          localStorage.removeItem('erp_user_email');
          currentUser = null;
          currentRole = null;
          showLogin();
          document.getElementById('loginUsername').value = '';
          document.getElementById('loginPassword').value = '';
          loadUsernameDropdown();
        }
              
        function checkSession() {
          const savedUser  = localStorage.getItem('erp_logged_user');
          const savedRole  = localStorage.getItem('erp_logged_role');
          const savedToken = localStorage.getItem('erp_session_token');
          if (savedUser && savedRole && savedToken) {
            google.script.run
              .withSuccessHandler(function(res) {
                if (res.valid) {
                  currentUser = savedUser;
                  currentRole = savedRole;
                  document.getElementById('loginContainer').classList.add('hidden');
                  document.getElementById('appContainer').classList.remove('hidden');
                  document.getElementById('profileUserName').innerText = savedUser;
                  document.getElementById('profileUserRole').innerText = savedRole;
                  loadPageContent('delegation');
                } else {
                  // Token invalid — clear and show login
                  logout();
                }
              })
              .withFailureHandler(function() { logout(); })
              .validateSession(savedUser, savedToken);
          } else {
            showLogin();
            loadUsernameDropdown();
          }
        }

        document.getElementById('sendOtpBtn').addEventListener('click', () => {
          const uname = document.getElementById('resetUsername').value.trim();
          const msg   = document.getElementById('resetMsg');
          if (!uname) { 
            msg.textContent = 'Please enter your username.'; 
            msg.className = 'text-sm text-center text-red-500'; 
            msg.classList.remove('hidden'); 
            return; 
          }
          
          document.getElementById('sendOtpBtn').disabled = true;
          document.getElementById('sendOtpBtn').textContent = 'Sending…';
          
          google.script.run
            .withSuccessHandler(res => {
              document.getElementById('sendOtpBtn').disabled = false;
              document.getElementById('sendOtpBtn').textContent = 'Send OTP';
              msg.classList.remove('hidden');
              if (res.success) {
                msg.textContent = res.message;
                msg.className = 'text-sm text-center text-green-600';
                document.getElementById('resetStep1').classList.add('hidden');
                document.getElementById('resetStep2').classList.remove('hidden');
              } else {
                msg.textContent = res.message;
                msg.className = 'text-sm text-center text-red-500';
              }
            })
            .withFailureHandler(err => {
              // Prevents the button from being stuck on "Sending..." if there is a severe backend error
              document.getElementById('sendOtpBtn').disabled = false;
              document.getElementById('sendOtpBtn').textContent = 'Send OTP';
              msg.classList.remove('hidden');
              msg.textContent = 'Backend Error: ' + err.message;
              msg.className = 'text-sm text-center text-red-500';
            })
            .initiatePasswordReset(uname);
        });
        document.getElementById('confirmResetBtn').addEventListener('click', () => {
          const uname   = document.getElementById('resetUsername').value.trim();
          const otp     = document.getElementById('resetOtp').value.trim();
          const newPass = document.getElementById('resetNewPass').value;
          const msg     = document.getElementById('resetMsg');
          if (!otp || !newPass) { msg.textContent = 'Please fill both fields.'; msg.className = 'text-sm text-center text-red-500'; msg.classList.remove('hidden'); return; }
          document.getElementById('confirmResetBtn').disabled = true;
          document.getElementById('confirmResetBtn').textContent = 'Resetting…';
          google.script.run.withSuccessHandler(res => {
            document.getElementById('confirmResetBtn').disabled = false;
            document.getElementById('confirmResetBtn').textContent = 'Reset Password';
            msg.classList.remove('hidden');
            msg.textContent = res.message;
            msg.className = `text-sm text-center ${res.success ? 'text-green-600' : 'text-red-500'}`;
            if (res.success) setTimeout(() => document.getElementById('resetModal').classList.add('hidden'), 2000);
          }).confirmPasswordReset(uname, otp, newPass);
        });

         function configureNavigationByRole(role, department) {
          const normRole = (role || '').trim().toLowerCase();
          const normDept = (department || '').trim().toLowerCase();

          const navDelegation = document.getElementById('navDelegation');
          const navChecklist  = document.getElementById('navChecklist');
          const navTasks      = document.getElementById('navTasks');

          // Pehle sabko hide kardo (To avoid confusion on login)
          if (navDelegation) navDelegation.classList.add('hidden');
          if (navChecklist)  navChecklist.classList.add('hidden');
          if (navTasks)      navTasks.classList.add('hidden');

          // Condition 1: Delegator + Admin -> All 3 enabled
          if (normRole === 'delegator' && normDept === 'admin') {
            if (navDelegation) navDelegation.classList.remove('hidden');
            if (navChecklist)  navChecklist.classList.remove('hidden');
            if (navTasks)      navTasks.classList.remove('hidden');
            loadPageContent('delegation');
          } 
          // Condition 2: Delegator + Not Admin -> Only Task & Delegation
          else if (normRole === 'delegator' && normDept !== 'admin') {
            if (navDelegation) navDelegation.classList.remove('hidden');
            if (navTasks)      navTasks.classList.remove('hidden');
            loadPageContent('delegation');
          } 
          // Condition 3: Delegatee + Not Admin -> Only Task
          else if (normRole === 'delegatee' && normDept !== 'admin') {
            if (navTasks)      navTasks.classList.remove('hidden');
            loadPageContent('tasks');
          }
          // Admin ya PC role direct fallback
          else if (normRole === 'admin' || normRole === 'pc') {
            if (navDelegation) navDelegation.classList.remove('hidden');
            if (navChecklist)  navChecklist.classList.remove('hidden');
            if (navTasks)      navTasks.classList.remove('hidden');
            loadPageContent('delegation');
          }
          else {
            if (navTasks) navTasks.classList.remove('hidden');
            loadPageContent('tasks');
          }
        }
